def defaults():

def write_ini(path, p, fit=None, section="hypscat"):
    """Write a parameter file (same key=value layout as CrystalScatter ``*.ini``, ``FITFLAG_<key>=NN``)."""
    fit = fit or {}
    with open(path, "w", encoding="utf-8") as f:
        f.write("[%s]\n" % quote(section))
        for k, v in p.items():
            if isinstance(v, bool):
                v = "true" if v else "false"
            f.write("%s=%s\n" % (k, v))
            if k in NUM:
                f.write("FITFLAG_%s=%s\n" % (k, "10" if fit.get(k) else "00"))

def read_ini(path):
    """Read an .ini written by this GUI or by SasScatter2.  Returns (params, fitflags, section)."""
    p, fit, section = defaults(), {}, ""
    with open(path, encoding="utf-8", errors="replace") as f:
        for line in f:
            line = line.strip()
            if not line or line.startswith(";"):
                continue
            if line.startswith("["):
                section = unquote(line.strip("[]"))
                continue
            if "=" not in line:
                continue
            k, v = line.split("=", 1)
            if k.startswith("FITFLAG_"):
                fit[k[8:]] = v[:1] == "1"
                continue
            k = {"Alfa": "Alpha", "CalcQmax": None}.get(k, k)
            if k is None:
                continue
            if k in BOOLS:
                p[k] = v.strip().lower() in ("true", "1")
            elif k in COMBOS or k in ("QuadrantFactor", "Dimension"):     # (also the pySCAtter combos)
                p[k] = int(float(v))
            elif k in NUM:
                p[k] = float(v)
            elif k == "Comment":
                p[k] = v
    return p, fit, section

def relevant(p):
    """Numeric parameters that the calculation uses with the combo settings of ``p``."""
    kind, inter = int(p["ComboBoxParticle"]), int(p["ComboBoxInterior"])
    lat = LTYPE_KIND.get(int(p["LType"])) is not None
    ordis = int(p["Ordis"])
    used = {"I0", "Base", "EditQmax", "EditQmin", "EditQsteps", "Editqz", "EditRadius"}
    if p.get("EditQmaxPreset", True):        # 2D: square q map of +-q max
        used.add("GridPoints")
    else:                                    # 2D: pixels of the detector
        used |= {"EditWavelength", "EditDet", "EditPixelX", "EditPixelY", "EditPixelNoX", "EditPixelNoY",
                 "BeamPosX", "BeamPosY"}
    # particle: the same order of precedence as engine.make_particle
    if kind in (9, 10, 11):                      # chains, Teubner-Strey: no interior
        used |= {9: {"EditSigma"}, 10: {"Length", "SigmaL"}, 11: {"Length"}}[kind]
    elif inter in (3, 4) and kind in (0, 3):     # multi-shell / myelin
        used |= {"EditSigma", "ShellN", "ShellThick"}
        used |= {"EditRho", "ShellRho2"} if inter == 3 else {"EditRadiusi"}   # myelin: one contrast (a scale)
    elif kind == 3:                              # vesicle: inner radius R, outer radius L
        used |= {"EditSigma", "Length", "SigmaL"}
    elif inter in (1, 2) and kind in (0, 1, 2, 5):
        used |= {"EditSigma", "EditRadiusi", "EditRho"}
        if kind != 0:
            used.add("Length")
        elif inter == 2 and int(p.get("ShellProfile", 0)) == 1:
            used.add("ProfileAlpha")
    else:
        used.add("EditSigma")
        if kind in (1, 2, 5, 6, 7):
            used.add("Length")
        if kind in (1, 2):
            used.add("SigmaL")
        if kind == 6:
            used.add("EditRadiusi")
        if kind in (7, 8):
            used.add("Alpha")
    if ordis != 7 and kind in _ANISO:
        if ordis != 13:                      # fibre: uniform about the beam, no director
            used |= {"theta", "phi"}
        if ordis not in (6, 13):
            used.add("EditDbeta")
    if lat:
        used |= {"EditDomainSize", "EditDebyeWaller", "HKLmax"}
        used |= CELL_KEYS.get(LTYPE_KIND[int(p["LType"])], _CELL_ALL)
        if int(p["ComboBoxPeak"]) in (4, 5, 6):
            used.add("EditPeakPar")
        if int(p["ComboBoxPeak"]) == 7 and ordis == 7:            # anisotropic Gaussian: domain sizes per axis
            used |= {"SigX", "SigY", "SigZ"}
            used.discard("EditDomainSize")
        elif not (p.get("RadButDebyeScherrer", True) or p.get("RadioButtonPara", False)):
            used.discard("EditDomainSize")                         # fixed width 1, as in CrystalScatter
        if ordis != 7:
            used |= {"ucpsi", "ucn1", "ucn2", "ucn3"}
            if int(p["ComboBoxPeak"]) <= 5 and ordis != 13:          # CrystalScatter form: azimuthal spread
                used.add("EditAzi")
    elif int(p.get("StructureFactor", 0)) > 0:
        from .models_ext import STRUCTURE_FACTORS
        used |= set(STRUCTURE_FACTORS[int(p["StructureFactor"])][2])
    return used
