Source code for pysasfit.core.background

r"""Background models added to the particle scattering.

    I_model(q) = (smeared) c(q) F(q)^2  +  B(q; theta)

The background parameters theta can be linear (constant, amplitudes) or
nonlinear (power-law exponent, correlation length). They are not
regularised: they are extra columns of the Gauss-Newton Jacobian with a flat
prior and are integrated out in the Laplace approximation of the evidence,
exactly like the flat background of Hansen (2016).

Positive parameters (amplitudes, exponents, lengths) are fitted as
z = ln(theta), which keeps them positive and makes the Gaussian approximation
better behaved. The background is evaluated at the measured q (not smeared).

Built-in terms
--------------

::

    Constant          c
    PowerLaw          A q^(-n)                 (low-q upturn of large aggregates, Porod n = 4,
                                                mass fractals 1 < n < 3, surface fractals 3 < n < 4)
    OrnsteinZernike   I_L / (1 + (q xi)^2)     (concentration fluctuations / network blobs)
    FunctionTerm      any user function f(q, *theta) with a numerical Jacobian

Example::

    bg = Background([Constant(), PowerLaw(n=4.0, fit_n=False)])
    fit = DensityProfileFit(q, I, s, background=bg)
"""

    def grad(self, q, th):
        """d f / d theta (len(q) x k); default: central differences."""
        th = np.asarray(th, float)
        out = np.empty((q.size, th.size))
        for j in range(th.size):
            h = 1e-6 * max(abs(th[j]), 1e-8)
            tp, tm = th.copy(), th.copy()
            tp[j] += h
            tm[j] -= h
            out[:, j] = (self.f(q, tp) - self.f(q, tm)) / (2 * h)
        return out

#Start values of the fitted background terms (pySASfit preferences:
#numerics.background.*). The fit is a local Levenberg-Marquardt, so the start
#decides which minimum is found when the background and the particle term
#compete. constant_level x median of the highest-q constant_tail fraction of
#the points; power law: amplitude powerlaw_level x I(q_min) q_min^n;
#Ornstein-Zernike: I_L = oz_level x max I, xi = 1/median q.
START_DEFAULTS = dict(constant_level=0.5, constant_tail=0.05, powerlaw_level=0.05, oz_level=0.05)

        out[:, 1] = -th[0] * 2.0 * q**2 * th[1] / d**2

class FunctionTerm(BackgroundTerm):

class Background:
    """Sum of background terms with a common parameter vector.

    Internally the free parameters are packed into z (z = ln theta for positive
    ones); fixed parameters keep their values.
    """

    def __init__(self, terms=()):
        self.terms = list(terms)
        self._ready = False

    def setup(self, q, I):
        """Fill unspecified starting values from the data."""
        q = np.asarray(q, float)
        I = np.asarray(I, float)
        for t in self.terms:
            g = t.guess(q, I)
            t.values = [g[j] if v is None else v for j, v in enumerate(t.values)]
        self._ready = True
        return self



    def _flat(self):
        # cached: called several times per model evaluation (2026-10-21, ~25 % of
        # the run time of a density-profile fit); the key detects any change of
        # the terms' values or free/positive flags
        key = tuple((id(t), tuple(t.values), tuple(t.free), tuple(t.positive)) for t in self.terms)
        c = getattr(self, "_flat_cache", None)
        if c is not None and c[0] == key:
            return c[1]
        out = self._flat_build()
        self._flat_cache = (key, out)
        return out

    def _struct(self):
        """Index arrays derived from _flat (cached with it): free indices, which free ones are positive, term
        slices, and whether every term is a Constant (the common flat background: value/jac without the loop)."""
        flat = self._flat()
        c = getattr(self, "_struct_cache", None)
        if c is not None and c[0] is flat:
            return c[1]
        _, v, fr, pos, _ = flat
        slices, i = [], 0
        for t in self.terms:
            k = len(t.names)
            slices.append((t, i, i + k))
            i += k
        st = dict(v=v, fr_idx=np.flatnonzero(fr), pfree=pos[fr], any_pos=bool(pos[fr].any()),
                  slices=slices, all_const=bool(self.terms) and all(type(t) is Constant for t in self.terms))
        self._struct_cache = (flat, st)
        return st

    def _flat_build(self):
        names, vals, free, pos, owner = [], [], [], [], []
        for it, t in enumerate(self.terms):
            for j, nm in enumerate(t.names):
                names.append(f"{type(t).__name__}.{nm}" if not isinstance(t, FunctionTerm) else nm)
                vals.append(float(t.values[j]))
                free.append(bool(t.free[j]))
                pos.append(bool(t.positive[j]))
                owner.append((it, j))
        return names, np.array(vals), np.array(free, bool), np.array(pos, bool), owner

    @property
    def names(self):
        return self._flat()[0]

    @property
    def free_names(self):
        n, _, fr, _, _ = self._flat()
        return [a for a, f in zip(n, fr) if f]

    @property
    def n_free(self):
        return int(self._flat()[2].sum())

    def z0(self):
        _, v, fr, pos, _ = self._flat()
        z = v[fr].copy()
        p = pos[fr]
        z[p] = np.log(np.maximum(z[p], 1e-300))
        return z

    def theta(self, z):
        """Full parameter vector for packed free parameters z."""
        st = self._struct()
        th = st["v"].copy()
        zf = np.asarray(z, float)
        if st["any_pos"]:
            zf = zf.copy()
            p = st["pfree"]
            zf[p] = np.exp(np.clip(zf[p], -700, 700))
        th[st["fr_idx"]] = zf
        return th

    def value(self, q, z=None):
        z = self.z0() if z is None else z
        th = self.theta(z)
        st = self._struct()
        q = np.asarray(q, float)
        if st["all_const"]:                       # flat background: same sum as the loop below
            c = 0.0
            for x in th:
                c = c + x
            return np.full(q.shape, c)
        out = np.zeros_like(q)
        for t, i, j in st["slices"]:
            out = out + t.f(q, th[i:j])
        return out

    def jac(self, q, z=None):
        """d B / d z (len(q) x n_free)."""
        st = self._struct()
        if st["all_const"]:                       # d c / d c = 1 for every free constant
            return np.ones((np.asarray(q).size, st["fr_idx"].size))
        z = self.z0() if z is None else z
        th = self.theta(z)
        _, _, fr, pos, _ = self._flat()
        out = np.empty((np.asarray(q).size, st["fr_idx"].size))
        c = 0
        for t, i, k1 in st["slices"]:
            if fr[i:k1].any():
                g = t.grad(q, th[i:k1])
                for j in range(k1 - i):
                    if fr[i + j]:
                        out[:, c] = g[:, j] * th[i + j] if pos[i + j] else g[:, j]   # d/dz = theta d/dtheta
                        c += 1
        return out

    def constant(self, z=None):
        """Sum of the Constant terms (the 'flat background')."""
        th = self.theta(self.z0() if z is None else z)
        c, i = 0.0, 0
        for t in self.terms:
            k = len(t.names)
            if isinstance(t, Constant):
                c += th[i]
            i += k
        return c

    def describe(self, z=None):
        th = self.theta(self.z0() if z is None else z)
        names, _, fr, _, _ = self._flat()
        return {n: (float(v), bool(f)) for n, v, f in zip(names, th, fr)}

    def to_free_theta(self, z):
        """Free parameters in natural units (for error propagation)."""
        _, _, fr, _, _ = self._flat()
        return self.theta(z)[fr]
